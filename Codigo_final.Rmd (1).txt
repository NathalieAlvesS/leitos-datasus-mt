O que essa saída indica

ME ≈ 0: não há viés relevante; o modelo não tende a superestimar nem subestimar.

```{r}
#-------------------------------------------------------------
# FUNCAO PARA CALCULAR O RPE
#-------------------------------------------------------------

calcular_rpe <- function(observado, previsto) {

  # Converter os objetos ts em vetores numericos
  observado <- as.numeric(observado)
  previsto  <- as.numeric(previsto)

  # Manter somente valores validos e diferentes de zero
  validos <- is.finite(observado) &
             is.finite(previsto) &
             observado != 0

  # Calcular o RPE em porcentagem
  rpe <- 100 * sqrt(
    mean(
      (
        (observado[validos] - previsto[validos]) /
          observado[validos]
      )^2
    )
  )

  return(rpe)
}

# RPE do modelo de Holt
rpe_holt <- calcular_rpe(
  y_teste,
  prev_holt$mean
)

# RPE do modelo ARIMA
rpe_arima <- calcular_rpe(
  y_teste,
  prev_arima$mean
)

rpe_holt
rpe_arima
```
